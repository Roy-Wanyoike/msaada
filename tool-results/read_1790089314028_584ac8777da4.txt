    81→/**
    82→ * County Triage Dashboard.
    83→ *
    84→ * Reads ONLY from `/api/dashboard`, which is an aggregate-only endpoint.
    85→ * The data-access layer (`getDashboardStats`) never selects indicator text,
    86→ * so this view is de-identified by construction.
    87→ *
    88→ * On first mount, if `totals.total === 0`, we auto-call `POST /api/seed`
    89→ * (idempotent) so judges never see an empty dashboard.
    90→ */
    91→export default function DashboardPage() {
    92→  const { toast } = useToast();
    93→  const [stats, setStats] = useState<DashboardStats | null>(null);
    94→  const [audit, setAudit] = useState<DashboardPayload["audit"]>([]);
    95→  const [followUpStats, setFollowUpStats] = useState<DashboardPayload["followUpStats"] | null>(null);
    96→  const [lastUpdated, setLastUpdated] = useState<number | null>(null);
    97→  const [scope, setScope] = useState<DashboardPayload["scope"]>({
    98→    county: null,
    99→    mode: "all",
   100→  });
   101→  const [chvCounty, setChvCounty] = useState<string | null>(null);
   102→  const [scopeMode, setScopeMode] = useState<"mine" | "all">("all");
   103→  const [state, setState] = useState<LoadState>("loading");
   104→  const [errorMsg, setErrorMsg] = useState<string | null>(null);
   105→  const [seeding, setSeeding] = useState(false);
   106→  const [refreshing, setRefreshing] = useState(false);
   107→  const [days, setDays] = useState<7 | 14 | 30>(14);
   108→  /** True once we've attempted the auto-seed (so we don't loop on failure). */
   109→  const autoSeedTriedRef = useRef(false);
   110→  /** Monotonic request counter to race-guard stale fetches. */
   111→  const reqIdRef = useRef(0);
   112→
   113→  const loadStats = useCallback(
   114→    async (opts?: {
   115→      silent?: boolean;
   116→      days?: number;
   117→      scope?: "mine" | "all";
   118→    }): Promise<DashboardStats | null> => {
   119→      const reqId = ++reqIdRef.current;
   120→      const rangeDays = opts?.days ?? days;
   121→      const scopeQ = opts?.scope ?? scopeMode;
   122→      if (!opts?.silent) setState((s) => (s === "ready" ? s : "loading"));
   123→      try {
   124→        const res = await fetch(
   125→          `/api/dashboard?days=${rangeDays}&scope=${scopeQ}`,
   126→          { cache: "no-store" }
   127→        );
   128→        if (!res.ok) {
   129→          throw new Error(`Dashboard endpoint returned HTTP ${res.status}`);
   130→        }
   131→        const data = (await res.json()) as DashboardPayload;
   132→        // Drop stale responses (a newer refresh superseded us).
   133→        if (reqId !== reqIdRef.current) return null;
   134→        setStats({
   135→          byCounty: data.byCounty,
   136→          byDay: data.byDay,
   137→          byTag: data.byTag,
   138→          totals: data.totals,
   139→        });
   140→        setAudit(data.audit ?? []);
   141→        setFollowUpStats(data.followUpStats ?? null);
   142→        setScope(data.scope ?? { county: null, mode: "all" });
   143→        setLastUpdated(Date.now());
   144→        setState("ready");
   145→        setErrorMsg(null);
   146→        return data;
   147→      } catch (err) {
   148→        if (reqId !== reqIdRef.current) return null;
   149→        const msg =
   150→          err instanceof Error ? err.message : "Failed to load dashboard";
   151→        setErrorMsg(msg);
   152→        setState("error");
   153→        if (!opts?.silent) {
   154→          toast({
   155→            title: "Couldn't load dashboard",
   156→            description: msg,
   157→            variant: "destructive",
   158→          });
   159→        }
   160→        return null;
   161→      }
   162→    },
   163→    [days, scopeMode, toast]
   164→  );
   165→
   166→  const seedDemo = useCallback(
   167→    async (opts?: { silent?: boolean }): Promise<void> => {
   168→      setSeeding(true);
   169→      try {
   170→        const res = await fetch("/api/seed", {
   171→          method: "POST",
   172→          headers: { "Content-Type": "application/json" },
   173→        });
   174→        if (!res.ok) {
   175→          throw new Error(`Seed endpoint returned HTTP ${res.status}`);
   176→        }
   177→        const body = (await res.json()) as { seeded: number };
   178→        if (!opts?.silent) {
   179→          toast({
   180→            title: `Seeded ${body.seeded} demo observation${
   181→              body.seeded === 1 ? "" : "s"
   182→            }`,
   183→            description: "Charts will refresh in a moment.",
   184→          });
   185→        }
   186→        await loadStats({ silent: true });
   187→      } catch (err) {
   188→        const msg =
   189→          err instanceof Error ? err.message : "Failed to seed demo data";
   190→        toast({
   191→          title: "Seed failed",
   192→          description: msg,
   193→          variant: "destructive",
   194→        });
   195→      } finally {
   196→        setSeeding(false);
   197→      }
   198→    },
   199→    [loadStats, toast]
   200→  );
   201→
   202→  // On mount, also hydrate the session CHV (for the RBAC county-scope).
   203→  useEffect(() => {
   204→    let cancelled = false;
   205→    (async () => {
   206→      try {
   207→        const res = await fetch("/api/auth/me", { credentials: "same-origin" });
   208→        const data = (await res.json()) as { chv: { county: string } | null };
   209→        if (cancelled) return;
   210→        if (data.chv?.county) {
   211→          setChvCounty(data.chv.county);
   212→          // Default to the CHV's own county scope (RBAC default).
   213→          setScopeMode("mine");
   214→        }
   215→      } catch {
   216→        // No session — fine, dashboard stays in all-county demo mode.
   217→      }
   218→    })();
   219→    return () => {
   220→      cancelled = true;
   221→    };
   222→  }, []);
   223→
   224→  // Initial mount: load, then auto-seed if empty (idempotent — only once).
   225→  useEffect(() => {
   226→    let cancelled = false;
   227→    (async () => {
   228→      const data = await loadStats({ silent: true });
   229→      if (cancelled || !data) return;
   230→      if (data.totals.total === 0 && !autoSeedTriedRef.current) {
   231→        autoSeedTriedRef.current = true;
   232→        await seedDemo({ silent: true });
   233→      }
   234→    })();
   235→    return () => {
   236→      cancelled = true;
   237→    };
   238→  }, [loadStats, seedDemo]);
   239→
   240→  const handleRefresh = useCallback(async () => {
   241→    setRefreshing(true);
   242→    await loadStats();
   243→    setRefreshing(false);
   244→  }, [loadStats]);
   245→
   246→  // Re-fetch when the time-range filter or the RBAC scope changes.
   247→  useEffect(() => {
   248→    void loadStats({ silent: true });
   249→  }, [days, scopeMode, loadStats]);
   250→
   251→  const handleDaysChange = useCallback((next: 7 | 14 | 30) => {
   252→    setDays(next);
   253→  }, []);
   254→
   255→  const handleScopeChange = useCallback((next: "mine" | "all") => {
   256→    setScopeMode(next);
   257→  }, []);
   258→
   259→  /** CSV export of the county table — de-identified (counts only). */
   260→  const handleExportCsv = useCallback(() => {
   261→    if (!stats) return;
   262→    const rows = stats.byCounty;
   263→    const header = [
   264→      "County",
   265→      "Total",
   266→      "Routine",
   267→      "Needs Follow-up",
   268→      "Needs Facility Referral",
   269→      "Escalation",
   270→    ];
   271→    const lines = [header.join(",")];
   272→    for (const r of rows) {
   273→      lines.push(
   274→        [
   275→          r.county,
   276→          r.total,
   277→          r.routine,
   278→          r.needs_followup,
   279→          r.needs_facility_referral,
   280→          r.escalation,
   281→        ].join(",")
   282→      );
   283→    }
   284→    const csv = lines.join("\n");
   285→    const blob = new Blob([csv], { type: "text/csv;charset=utf-8;" });
   286→    const url = URL.createObjectURL(blob);
   287→    const a = document.createElement("a");
   288→    a.href = url;
   289→    a.download = `msaada-county-triage-${days}d-${new Date()
   290→      .toISOString()
   291→      .slice(0, 10)}.csv`;
   292→    document.body.appendChild(a);
   293→    a.click();
   294→    document.body.removeChild(a);
   295→    URL.revokeObjectURL(url);
   296→    toast({
   297→      title: "CSV exported",
   298→      description: `${rows.length} counties · last ${days} days`,
   299→    });
   300→  }, [stats, days, toast]);
   301→
   302→  // ----- Derived view state -----
   303→  const isEmpty =
   304→    state === "ready" && stats !== null && stats.totals.total === 0;
   305→
   306→  return (
   307→    <div className="flex min-h-screen flex-col bg-background">
   308→      <AppNav />
   309→      <main className="flex-1">
   310→        <div className="mx-auto w-full max-w-7xl px-4 py-6 sm:px-6 sm:py-8 lg:px-8">
   311→          <DashboardHeader
   312→            onRefresh={handleRefresh}
   313→            refreshing={refreshing}
   314→            days={days}
   315→            onDaysChange={handleDaysChange}
   316→            onExportCsv={handleExportCsv}
   317→            chvCounty={chvCounty}
   318→            scopeMode={scopeMode}
   319→            onScopeChange={handleScopeChange}
   320→            lastUpdated={lastUpdated}
   321→          />
   322→
   323→          {state === "loading" ? (
   324→            <DashboardSkeleton />
   325→          ) : state === "error" ? (
   326→            <ErrorState
   327→              message={errorMsg ?? "Unknown error"}
   328→              onRetry={() => {
   329→                void loadStats();
   330→              }}
   331→            />
   332→          ) : !stats || isEmpty ? (
   333→            <EmptyState
   334→              onSeed={() => {
   335→                void seedDemo();
   336→              }}
   337→              seeding={seeding}
   338→            />
   339→          ) : (
   340→            <DashboardView
   341→              stats={stats}
   342→              audit={audit}
   343→              scope={scope}
   344→              followUpStats={followUpStats}
   345→              onSeed={() => {
   346→                void seedDemo();
   347→              }}
   348→              seeding={seeding}
   349→            />
   350→          )}
   351→        </div>
   352→      </main>
   353→
   354→      <DashboardFooter />
   355→    </div>
   356→  );
   357→}

   541→/* ------------------------------------------------------------------ */
   542→/* Main dashboard view (data loaded & non-empty)                       */
   543→/* ------------------------------------------------------------------ */
   544→
   545→function DashboardView({
   546→  stats,
   547→  audit,
   548→  scope,
   549→  followUpStats,
   550→  onSeed,
   551→  seeding,
   552→}: {
   553→  stats: DashboardStats;
   554→  audit: DashboardPayload["audit"];
   555→  scope: DashboardPayload["scope"];
   556→  followUpStats: DashboardPayload["followUpStats"] | null;
   557→  onSeed: () => void;
   558→  seeding: boolean;
   559→}) {
   560→  const { totals } = stats;
   561→  // Defensive guards — if the API returns a partial payload during a scope
   562→  // switch or stale fetch, fall back to empty arrays so the charts / helpers
   563→  // never receive undefined (was crashing with "byDay is not iterable").
   564→  const guardedStats: DashboardStats = {
   565→    byCounty: Array.isArray(stats.byCounty) ? stats.byCounty : [],
   566→    byDay: Array.isArray(stats.byDay) ? stats.byDay : [],
   567→    byTag: Array.isArray(stats.byTag) ? stats.byTag : [],
   568→    totals,
   569→  };
   570→
   571→  // KPI hints: small trend hints derivable from the regional byDay series.
   572→  const followupDelta = weeklyDelta(guardedStats.byDay, "needs_followup");
   573→  const escalationDelta = weeklyDelta(guardedStats.byDay, "escalation");
   574→
   575→  const followupHint = `${pct(totals.needs_followup, totals.total)}% of total${
   576→    followupDelta.noBaseline
   577→      ? ""
   578→      : ` · ${followupDelta.delta >= 0 ? "+" : ""}${followupDelta.delta} vs prev 7d`
   579→  }`;
   580→  const escalationHint = `${pct(totals.escalation, totals.total)}% of total${
   581→    escalationDelta.noBaseline
   582→      ? ""

   801→/* Freshness badge — shows "updated Xm ago" + staleness color.         */
   802→/* ------------------------------------------------------------------ */
   803→
   804→function FreshnessBadge({ lastUpdated }: { lastUpdated: number }) {
   805→  // Tick every 30s so the relative time stays fresh.
   806→  const [, setNow] = useState(Date.now());
   807→  useEffect(() => {
   808→    const id = setInterval(() => setNow(Date.now()), 30000);
   809→    return () => clearInterval(id);
   810→  }, []);
   811→
   812→  const ageMs = Date.now() - lastUpdated;
   813→  const ageMin = Math.floor(ageMs / 60000);
   814→  const label =
   815→    ageMin < 1 ? "just now" : ageMin < 60 ? `${ageMin}m ago` : `${Math.floor(ageMin / 60)}h ago`;
   816→  // Stale = >5 min. Fresh = <1 min.
   817→  const stale = ageMin > 5;
   818→  const fresh = ageMin < 1;
   819→  const cls = fresh
   820→    ? "border-emerald-200 bg-emerald-50 text-emerald-700 dark:border-emerald-900 dark:bg-emerald-950/40 dark:text-emerald-300"
   821→    : stale
   822→      ? "border-amber-200 bg-amber-50 text-amber-700 dark:border-amber-900 dark:bg-amber-950/40 dark:text-amber-300"
   823→      : "border-border bg-muted/40 text-muted-foreground";
   824→
   825→  return (
   826→    <span
   827→      className={`inline-flex items-center gap-1 rounded-full border px-2 py-0.5 text-[10px] font-medium ${cls}`}
   828→      title={`Data last refreshed ${label}`}
   829→    >
   830→      <span
   831→        className={`inline-block size-1.5 rounded-full ${
   832→          fresh ? "bg-emerald-500 animate-pulse" : stale ? "bg-amber-500" : "bg-muted-foreground"
   833→        }`}
   834→        aria-hidden
   835→      />
   836→      Updated {label}
   837→    </span>
   838→  );
   839→}
