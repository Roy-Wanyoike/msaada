     5→     5→/**
     6→     6→ * Data-access layer. Mirrors the Supabase RLS design:
     7→     7→ *  - insertTriageRecord enforces submittedById ownership (the RLS
     8→     8→ *    auth.uid() = submitting user equivalent).
     9→     9→ *  - The dashboard aggregate functions NEVER select observed_indicators /
    10→    10→ *    chpNextAction / confidenceNote text — only counts. This is the
    11→    11→ *    "Postgres VIEW exposing only aggregated counts" equivalent, enforced
    12→    12→ *    at the data-access layer because SQLite has no native RLS.
    13→    13→ *  - The raw free-text observation is never persisted anywhere.
    14→    14→ */
    15→    15→
    16→    16→export async function insertTriageRecord(args: {
    17→    17→  submittedById: string;
    18→    18→  county: string;
    19→    19→  ward?: string;
    20→    20→  output: TriageModelOutput;
    21→    21→  fallbackUsed: boolean;
    22→    22→}): Promise<TriageRecordDTO> {
    23→    23→  const { submittedById, county, ward, output, fallbackUsed } = args;
    24→    24→  const isCrisis = output.escalation === true;
    25→    25→
    26→    26→  const created = await db.triageRecord.create({
    27→    27→    data: {
    28→    28→      submittedById,
    29→    29→      county,
    30→    30→      ward: ward ?? null,
    31→    31→      classification: isCrisis ? "needs_facility_referral" : output.classification,
    32→    32→      escalation: isCrisis,
    33→    33→      observedIndicators: JSON.stringify(
    34→    34→        isCrisis ? [] : output.observed_indicators
    35→    35→      ),
    36→    36→      aggregateTag: isCrisis ? "crisis_self_harm" : output.aggregate_tag ?? null,
    37→    37→      chpNextAction: isCrisis ? null : (isCrisis ? null : output.chp_next_action),
    38→    38→      chpInstruction: isCrisis ? output.chp_instruction : null,
    39→    39→      crisisLine: isCrisis ? output.crisis_line : null,
    40→    40→      confidenceNote: isCrisis
    41→    41→        ? "Crisis override triggered"
    42→    42→        : fallbackUsed
    43→    43→          ? output.confidence_note
    44→    44→          : (output.confidence_note ?? null),
    45→    45→    },
    46→    46→  });
    47→    47→
    48→    48→  return toDTO(created, fallbackUsed);
    49→    49→}

    51→    51→function toDTO(
    52→    52→  row: {
    53→    53→    id: string;
    54→    54→    createdAt: Date;
    55→    55→    county: string;
    56→    56→    ward: string | null;
    57→    57→    classification: string;
    58→    58→    escalation: boolean;
    59→    59→    observedIndicators: string;
    60→    60→    aggregateTag: string | null;
    61→    61→    chpNextAction: string | null;
    62→    62→    chpInstruction: string | null;
    63→    63→    crisisLine: string | null;
    64→    64→    confidenceNote: string | null;
    65→    65→  },
    66→    66→  fallbackUsed: boolean
    67→    67→): TriageRecordDTO {
    68→    68→  let indicators: string[] = [];
    69→    69→  try {
    70→    70→    const parsed = JSON.parse(row.observedIndicators);
    71→    71→    if (Array.isArray(parsed)) indicators = parsed.filter((x) => typeof x === "string");
    72→    72→  } catch {
    73→    73→    indicators = [];
    74→    74→  }
    75→    75→  return {
    76→    76→    id: row.id,
    77→    77→    createdAt: row.createdAt.toISOString(),
    78→    78→    county: row.county,
    79→    79→    ward: row.ward,
    80→    80→    classification: row.classification as Classification,
    81→    81→    escalation: row.escalation,
    82→    82→    observedIndicators: indicators,
    83→    83→    aggregateTag: row.aggregateTag,
    84→    84→    chpNextAction: row.chpNextAction,
    85→    85→    chpInstruction: row.chpInstruction,
    86→    86→    crisisLine: row.crisisLine,
    87→    87→    confidenceNote: row.confidenceNote,
    88→    88→    fallbackUsed,
    89→    89→  };
    90→    90→}

   174→   174→/**
   175→   175→ * Create a follow-up for a triage record (called by /api/triage when the
   176→   176→ * classification is needs_followup or needs_facility_referral). Due in 48h.
   177→   177→ * Idempotent — if an open follow-up already exists for this triage record,
   178→   178→ * returns the existing one instead of creating a duplicate.
   179→   179→ */
   180→   180→export async function createFollowUp(args: {
   181→   181→  triageRecordId: string;
   182→   182→  chvId: string;
   183→   183→  dueInHours?: number;
   184→   184→}): Promise<FollowUpDTO | null> {
   185→   185→  const dueInHours = args.dueInHours ?? 48;
   186→   186→  // Idempotent: check for an existing pending follow-up for this record.
   187→   187→  const existing = await db.followUp.findFirst({
   188→   188→    where: { triageRecordId: args.triageRecordId, status: "pending" },
   189→   189→    include: { triageRecord: { select: { county: true, ward: true, classification: true, escalation: true, aggregateTag: true, chpNextAction: true } } },
   190→   190→  });
   191→   191→  if (existing) return toFollowUpDTO(existing);
   192→   192→
   193→   193→  const dueAt = new Date();
   194→   194→  dueAt.setHours(dueAt.getHours() + dueInHours);
   195→   195→
   196→   196→  const created = await db.followUp.create({
   197→   197→    data: {
   198→   198→      triageRecordId: args.triageRecordId,
   199→   199→      chvId: args.chvId,
   200→   200→      dueAt,
   201→   201→    },
   202→   202→    include: { triageRecord: { select: { county: true, ward: true, classification: true, escalation: true, aggregateTag: true, chpNextAction: true } } },
   203→   203→  });
   204→   204→  return toFollowUpDTO(created);
   205→   205→}

   227→   227→/** Resolve a follow-up (mark done/missed). Ownership-scoped. */
   228→   228→export async function resolveFollowUp(args: {
   229→   229→  followUpId: string;
   230→   230→  chvId: string;
   231→   231→  status: "done" | "missed";
   232→   232→  resolutionNote?: string;
   233→   233→}): Promise<FollowUpDTO | null> {
   234→   234→  const { followUpId, chvId, status, resolutionNote } = args;
   235→   235→  // Ownership check: only the assigned CHV can resolve.
   236→   236→  const existing = await db.followUp.findUnique({ where: { id: followUpId } });
   237→   237→  if (!existing || existing.chvId !== chvId) return null;
   238→   238→  if (existing.status !== "pending") return null; // already resolved
   239→   239→
   240→   240→  const updated = await db.followUp.update({
   241→   241→    where: { id: followUpId },
   242→   242→    data: {
   243→   243→      status,
   244→   244→      resolvedAt: new Date(),
   245→   245→      resolutionNote: resolutionNote?.trim() || null,
   246→   246→    },
   247→   247→    include: {
   248→   248→      triageRecord: { select: { county: true, ward: true, classification: true, escalation: true, aggregateTag: true, chpNextAction: true } },
   249→   249→    },
   250→   250→  });
   251→   251→  return toFollowUpDTO(updated);
   252→   252→}

   271→   271→/**
   272→   272→ * Returns a CHV's personal aggregate stats (ownership-scoped). Computed from
   273→   273→ * their own records only — the RLS `auth.uid() = submitted_by` equivalent.
   274→   274→ */
   275→   275→export async function getMyStats(submittedById: string): Promise<ChvStats> {
   276→   276→  const now = new Date();
   277→   277→  const last7Start = new Date(now);
   278→   278→  last7Start.setDate(last7Start.getDate() - 7);
   279→   279→  last7Start.setHours(0, 0, 0, 0);
   280→   280→  const prev7Start = new Date(last7Start);
   281→   281→  prev7Start.setDate(prev7Start.getDate() - 7);
   282→   282→
   283→   283→  const [groups, last7Rows, prev7Rows, firstRow, counties] = await Promise.all([
   284→   284→    db.triageRecord.groupBy({
   285→   285→      by: ["classification", "escalation"],
   286→   286→      _count: true,
   287→   287→      where: { submittedById },
   288→   288→    }),
   289→   289→    db.triageRecord.count({
   290→   290→      where: { submittedById, createdAt: { gte: last7Start } },
   291→   291→    }),
   292→   292→    db.triageRecord.count({
   293→   293→      where: {
   294→   294→        submittedById,
   295→   295→        createdAt: { gte: prev7Start, lt: last7Start },
   296→   296→      },
   297→   297→    }),
   298→   298→    db.triageRecord.findFirst({
   299→   299→      where: { submittedById },
   300→   300→      orderBy: { createdAt: "asc" },
   301→   301→      select: { createdAt: true },
   302→   302→    }),
   303→   303→    db.triageRecord.groupBy({
   304→   304→      by: ["county"],
   305→   305→      _count: true,
   306→   306→      where: { submittedById },
   307→   307→    }),
   308→   308→  ]);
   309→   309→
   310→   310→  const stats: ChvStats = {
   311→   311→    total: 0,
   312→   312→    routine: 0,
   313→   313→    needs_followup: 0,
   314→   314→    needs_facility_referral: 0,
   315→   315→    escalation: 0,
   316→   316→    last7d: last7Rows,
   317→   317→    prev7d: prev7Rows,
   318→   318→    firstSubmission: firstRow ? firstRow.createdAt.toISOString() : null,
   319→   319→    countiesCovered: counties.length,
   320→   320→  };
   321→   321→
   322→   322→  for (const g of groups) {
   323→   323→    stats.total += g._count;
   324→   324→    if (g.classification === "routine") stats.routine += g._count;
   325→   325→    if (g.classification === "needs_followup") stats.needs_followup += g._count;
   326→   326→    if (g.classification === "needs_facility_referral")
   327→   327→      stats.needs_facility_referral += g._count;
   328→   328→    if (g.escalation) stats.escalation += g._count;
   329→   329→  }
   330→   330→
   331→   331→  return stats;
   332→   332→}

   357→   357→/**
   358→   358→ * Aggregate by county over the last `days` days. Uses groupBy — never touches
   359→   359→ * observed_indicators text.
   360→   360→ */
   361→   361→export async function getAggregateByCounty(
   362→   362→  days?: number
   363→   363→): Promise<CountyAggregate[]> {
   364→   364→  const since =
   365→   365→    days && days > 0
   366→   366→      ? (() => {
   367→   367→          const d = new Date();
   368→   368→          d.setHours(0, 0, 0, 0);
   369→   369→          d.setDate(d.getDate() - (days - 1));
   370→   370→          return d;
   371→   371→        })()
   372→   372→      : undefined;
   373→   373→  const groups = await db.triageRecord.groupBy({
   374→   374→    by: ["county", "classification", "escalation"],
   375→   375→    _count: true,
   376→   376→    where: since ? { createdAt: { gte: since } } : undefined,
   377→   377→  });
   378→   378→
   379→   379→  const byCounty = new Map<string, CountyAggregate>();
   380→   380→  const ensure = (county: string): CountyAggregate => {
   381→   381→    let a = byCounty.get(county);
   382→   382→    if (!a) {
   383→   383→      a = {
   384→   384→        county,
   385→   385→        routine: 0,
   386→   386→        needs_followup: 0,
   387→   387→        needs_facility_referral: 0,
   388→   388→        escalation: 0,
   389→   389→        total: 0,
   390→   390→      };
   391→   391→      byCounty.set(county, a);
   392→   392→    }
   393→   393→    return a;
   394→   394→  };
   395→   395→
   396→   396→  for (const g of groups) {
   397→   397→    const a = ensure(g.county);
   398→   398→    a.total += g._count;
   399→   399→    if (g.classification === "routine") a.routine += g._count;
   400→   400→    if (g.classification === "needs_followup") a.needs_followup += g._count;
   401→   401→    if (g.classification === "needs_facility_referral") a.needs_facility_referral += g._count;
   402→   402→    if (g.escalation) a.escalation += g._count;
   403→   403→  }
   404→   404→  return Array.from(byCounty.values()).sort((a, b) => b.total - a.total);
   405→   405→}

   407→   407→/**
   408→   408→ * Aggregate by day (last N days). Group by date string.
   409→   409→ */
   410→   410→export async function getAggregateByDay(days = 14): Promise<DailyAggregate[]> {
   411→   411→  const since = new Date();
   412→   412→  since.setHours(0, 0, 0, 0);
   413→   413→  since.setDate(since.getDate() - (days - 1));
   414→   414→
   415→   415→  const rows = await db.triageRecord.findMany({
   416→   416→    where: { createdAt: { gte: since } },
   417→   417→    select: { createdAt: true, classification: true, escalation: true },
   418→   418→  });
   419→   419→
   420→   420→  const byDay = new Map<string, DailyAggregate>();
   421→   421→  for (let i = 0; i < days; i++) {
   422→   422→    const d = new Date(since);
   423→   423→    d.setDate(d.getDate() + i);
   424→   424→    const key = d.toISOString().slice(0, 10);
   425→   425→    byDay.set(key, {
   426→   426→      day: key,
   427→   427→      routine: 0,
   428→   428→      needs_followup: 0,
   429→   429→      needs_facility_referral: 0,
   430→   430→      escalation: 0,
   431→   431→      total: 0,
   432→   432→    });
   433→   433→  }
   434→   434→
   435→   435→  for (const r of rows) {
   436→   436→    const key = r.createdAt.toISOString().slice(0, 10);
   437→   437→    const a = byDay.get(key);
   438→   438→    if (!a) continue;
   439→   439→    a.total++;
   440→   440→    if (r.classification === "routine") a.routine++;
   441→   441→    if (r.classification === "needs_followup") a.needs_followup++;
   442→   442→    if (r.classification === "needs_facility_referral") a.needs_facility_referral++;
   443→   443→    if (r.escalation) a.escalation++;
   444→   444→  }
   445→   445→  return Array.from(byDay.values()).sort((a, b) => a.day.localeCompare(b.day));
   446→   446→}

   677→   677→export interface FollowUpStats {
   678→   678→  pending: number;
   679→   679→  done: number;
   680→   680→  missed: number;
   681→   681→  /** Overdue = pending AND past dueAt. */
   682→   682→  overdue: number;

   685→   685→  total: number;

   687→   687→
   688→   688→/**
   689→   689→ * Returns aggregate follow-up stats (de-identified — counts only). Optionally
   690→   690→ * filtered by county and/or a date window. Used by the dashboard's follow-up
   691→   691→ * KPI card and the supervisor roster.
   692→   692→ */
   693→   693→export async function getFollowUpStats(
   694→   694→  opts: { county?: string; days?: number } = {}
   695→   695→): Promise<FollowUpStats> {
   696→   696→  const since = opts.days
   697→   697→    ? (() => {
   698→   698→        const d = new Date();
   699→   699→        d.setHours(0, 0, 0, 0);
   700→   700→        d.setDate(d.getDate() - (opts.days! - 1));
   701→   701→        return d;
   702→   702→      })()
   703→   703→    : undefined;
   704→   704→
   705→   705→  // County filter requires joining through TriageRecord — use the relation.
   706→   706→  const where = {
   707→   707→    ...(since ? { createdAt: { gte: since } } : {}),
   708→   708→    ...(opts.county ? { triageRecord: { county: opts.county } } : {}),
   709→   709→  };
   710→   710→
   711→   711→  const [statusGroups, overdueCount, total] = await Promise.all([
   712→   712→    db.followUp.groupBy({
   713→   713→      by: ["status"],
   714→   714→      _count: true,
   715→   715→      where,
   716→   716→    }),
   717→   717→    db.followUp.count({
   718→   718→      where: {
   719→   719→        ...where,
   720→   720→        status: "pending",
   721→   721→        dueAt: { lt: new Date() },
   722→   722→      },
   723→   723→    }),
   724→   724→    db.followUp.count({ where }),
   725→   725→  ]);
   726→   726→
   727→   727→  const stats: FollowUpStats = {
   728→   728→    pending: 0,
   729→   729→    done: 0,
   730→   730→    missed: 0,
   731→   731→    overdue: overdueCount,
   732→   732→    total,
   733→   733→    completionRate: 0,
   734→   734→  };
   735→   735→  for (const g of statusGroups) {
   736→   736→    if (g.status === "pending") stats.pending = g._count;
   737→   737→    if (g.status === "done") stats.done = g._count;
   738→   738→    if (g.status === "missed") stats.missed = g._count;
   739→   739→  }
   740→   740→  const resolved = stats.done + stats.missed;
   741→   741→  stats.completionRate = resolved > 0 ? Math.round((stats.done / resolved) * 100) : 0;
   742→   742→  return stats;
   743→   743→}

   788→   788→  days = 14

   790→   790→  const since = (() => {
   791→   791→    const d = new Date();
   792→   792→    d.setHours(0, 0, 0, 0);
   793→   793→    d.setDate(d.getDate() - (days - 1));
   794→   794→    return d;
   795→   795→  })();
   796→   796→  const last7Start = (() => {
   797→   797→    const d = new Date();
   798→   798→    d.setHours(0, 0, 0, 0);
   799→   799→    d.setDate(d.getDate() - 7);
   800→   800→    return d;
   801→   801→  })();
   802→   802→
   803→   803→  const where = {
   804→   804→    createdAt: { gte: since },
   805→   805→    ...(county ? { county } : {}),
   806→   806→  };
   807→   807→
   808→   808→  // Group by submittedById + classification + escalation for per-CHV breakdowns.
   809→   809→  const [groups, chvs, last7Counts, lastSubs] = await Promise.all([
   810→   810→    db.triageRecord.groupBy({
   811→   811→      by: ["submittedById", "classification", "escalation"],
   812→   812→      _count: true,
   813→   813→      where,
   814→   814→    }),
   815→   815→    db.chvUser.findMany({

   817→   817→      select: { id: true, county: true, ward: true },
   818→   818→    }),
   819→   819→    db.triageRecord.groupBy({
   820→   820→      by: ["submittedById"],
   821→   821→      _count: true,
   822→   822→      where: { ...where, createdAt: { gte: last7Start } },
   823→   823→    }),
   824→   824→    db.triageRecord.findMany({
   825→   825→      where,
   826→   826→      distinct: ["submittedById"],
   827→   827→      orderBy: { createdAt: "desc" },

   833→   833→  const last7Map = new Map(last7Counts.map((g) => [g.submittedById, g._count]));
   834→   834→  const lastSubMap = new Map(lastSubs.map((r) => [r.submittedById, r.createdAt.toISOString()]));
   835→   835→

   840→   840→      const meta = chvMeta.get(id);
   841→   841→      r = {
   842→   842→        chvLabel: `chv·${id.slice(-4)}`,
   843→   843→        county: meta?.county ?? "—",
   844→   844→        ward: meta?.ward ?? null,
   845→   845→        total: 0,
   846→   846→        routine: 0,

   855→   855→    return r;
   856→   856→  };
   857→   857→
   858→   858→  let totalAll = 0;
   859→   859→  let totalEsc = 0;
   860→   860→  for (const g of groups) {

   865→   865→    if (g.classification === "needs_followup") r.needs_followup += g._count;
   866→   866→    if (g.classification === "needs_facility_referral") r.needs_facility_referral += g._count;

   870→   870→    }
   871→   871→  }
   872→   872→
   873→   873→  const rows = Array.from(byChv.values()).sort((a, b) => b.total - a.total);
   874→   874→
   875→   875→  return {
   876→   876→    rows,
   877→   877→    totals: { chvs: rows.length, total: totalAll, escalations: totalEsc },
   878→   878→  };
   879→   879→}

   887→   887→export async function getDashboardStatsForCounty(
   888→   888→  county: string,
   889→   889→  days = 14
   890→   890→): Promise<DashboardStats & { county: string }> {
   891→   891→  const since = (() => {
   892→   892→    const d = new Date();
   893→   893→    d.setHours(0, 0, 0, 0);
   894→   894→    d.setDate(d.getDate() - (days - 1));
   895→   895→    return d;
   896→   896→  })();
   897→   897→
   898→   898→  const where = {
   899→   899→    county,
   900→   900→    createdAt: { gte: since },
   901→   901→  };
   902→   902→
   903→   903→  const [countyGroups, dayRows, tagGroups, totalsGroups] = await Promise.all([
   904→   904→    db.triageRecord.groupBy({
   905→   905→      by: ["county", "classification", "escalation"],
   906→   906→      _count: true,
   907→   907→      where,
   908→   908→    }),
   909→   909→    db.triageRecord.findMany({
   910→   910→      where,
   911→   911→      select: { createdAt: true, classification: true, escalation: true },
   912→   912→    }),
   913→   913→    db.triageRecord.groupBy({
   914→   914→      by: ["aggregateTag"],
   915→   915→      _count: true,
   916→   916→      orderBy: { _count: { aggregateTag: "desc" } },
   917→   917→      take: 12,
   918→   918→      where,
   919→   919→    }),
   920→   920→    db.triageRecord.groupBy({
   921→   921→      by: ["classification", "escalation"],
   922→   922→      _count: true,
   923→   923→      where,
   924→   924→    }),
   925→   925→  ]);
   926→   926→
   927→   927→  // byCounty — single row for the scoped county
   928→   928→  const countyAgg: CountyAggregate = {
   929→   929→    county,
   930→   930→    routine: 0,
   931→   931→    needs_followup: 0,
   932→   932→    needs_facility_referral: 0,
   933→   933→    escalation: 0,
   934→   934→    total: 0,
   935→   935→  };
   936→   936→  for (const g of countyGroups) {
   937→   937→    countyAgg.total += g._count;
   938→   938→    if (g.classification === "routine") countyAgg.routine += g._count;
   939→   939→    if (g.classification === "needs_followup") countyAgg.needs_followup += g._count;
   940→   940→    if (g.classification === "needs_facility_referral")
   941→   941→      countyAgg.needs_facility_referral += g._count;
   942→   942→    if (g.escalation) countyAgg.escalation += g._count;
   943→   943→  }
   944→   944→  const byCounty = countyAgg.total > 0 ? [countyAgg] : [];
   945→   945→
   946→   946→  // byDay — full N-day ladder, populated from dayRows
   947→   947→  const byDay = new Map<string, DailyAggregate>();
   948→   948→  for (let i = 0; i < days; i++) {
   949→   949→    const d = new Date(since);
   950→   950→    d.setDate(d.getDate() + i);
   951→   951→    const key = d.toISOString().slice(0, 10);
   952→   952→    byDay.set(key, {
   953→   953→      day: key,
   954→   954→      routine: 0,
   955→   955→      needs_followup: 0,
   956→   956→      needs_facility_referral: 0,
   957→   957→      escalation: 0,
   958→   958→      total: 0,
   959→   959→    });

   989→   989→      totals.needs_facility_referral += g._count;
