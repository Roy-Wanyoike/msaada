     5→/**
     6→ * Data-access layer. Mirrors the Supabase RLS design:
     7→ *  - insertTriageRecord enforces submittedById ownership (the RLS
     8→ *    auth.uid() = submitting user equivalent).
     9→ *  - The dashboard aggregate functions NEVER select observed_indicators /
    10→ *    chpNextAction / confidenceNote text — only counts. This is the
    11→ *    "Postgres VIEW exposing only aggregated counts" equivalent, enforced
    12→ *    at the data-access layer because SQLite has no native RLS.
    13→ *  - The raw free-text observation is never persisted anywhere.
    14→ */
    15→
    16→export async function insertTriageRecord(args: {
    17→  submittedById: string;
    18→  county: string;
    19→  ward?: string;
    20→  output: TriageModelOutput;
    21→  fallbackUsed: boolean;
    22→}): Promise<TriageRecordDTO> {
    23→  const { submittedById, county, ward, output, fallbackUsed } = args;
    24→  const isCrisis = output.escalation === true;
    25→
    26→  const created = await db.triageRecord.create({
    27→    data: {
    28→      submittedById,
    29→      county,
    30→      ward: ward ?? null,
    31→      classification: isCrisis ? "needs_facility_referral" : output.classification,
    32→      escalation: isCrisis,
    33→      observedIndicators: JSON.stringify(
    34→        isCrisis ? [] : output.observed_indicators
    35→      ),
    36→      aggregateTag: isCrisis ? "crisis_self_harm" : output.aggregate_tag ?? null,
    37→      chpNextAction: isCrisis ? null : (isCrisis ? null : output.chp_next_action),
    38→      chpInstruction: isCrisis ? output.chp_instruction : null,
    39→      crisisLine: isCrisis ? output.crisis_line : null,
    40→      confidenceNote: isCrisis
    41→        ? "Crisis override triggered"
    42→        : fallbackUsed
    43→          ? output.confidence_note
    44→          : (output.confidence_note ?? null),
    45→    },
    46→  });
    47→
    48→  return toDTO(created, fallbackUsed);
    49→}

    51→function toDTO(
    52→  row: {
    53→    id: string;
    54→    createdAt: Date;
    55→    county: string;
    56→    ward: string | null;
    57→    classification: string;
    58→    escalation: boolean;
    59→    observedIndicators: string;
    60→    aggregateTag: string | null;
    61→    chpNextAction: string | null;
    62→    chpInstruction: string | null;
    63→    crisisLine: string | null;
    64→    confidenceNote: string | null;
    65→  },
    66→  fallbackUsed: boolean
    67→): TriageRecordDTO {
    68→  let indicators: string[] = [];
    69→  try {
    70→    const parsed = JSON.parse(row.observedIndicators);
    71→    if (Array.isArray(parsed)) indicators = parsed.filter((x) => typeof x === "string");
    72→  } catch {
    73→    indicators = [];
    74→  }
    75→  return {
    76→    id: row.id,
    77→    createdAt: row.createdAt.toISOString(),
    78→    county: row.county,
    79→    ward: row.ward,
    80→    classification: row.classification as Classification,
    81→    escalation: row.escalation,
    82→    observedIndicators: indicators,
    83→    aggregateTag: row.aggregateTag,
    84→    chpNextAction: row.chpNextAction,
    85→    chpInstruction: row.chpInstruction,
    86→    crisisLine: row.crisisLine,
    87→    confidenceNote: row.confidenceNote,
    88→    fallbackUsed,
    89→  };
    90→}

    95→ * structured output (including observed_indicators, since they already

   174→/**
   175→ * Create a follow-up for a triage record (called by /api/triage when the
   176→ * classification is needs_followup or needs_facility_referral). Due in 48h.
   177→ * Idempotent — if an open follow-up already exists for this triage record,
   178→ * returns the existing one instead of creating a duplicate.
   179→ */
   180→export async function createFollowUp(args: {
   181→  triageRecordId: string;
   182→  chvId: string;
   183→  dueInHours?: number;
   184→}): Promise<FollowUpDTO | null> {
   185→  const dueInHours = args.dueInHours ?? 48;
   186→  // Idempotent: check for an existing pending follow-up for this record.
   187→  const existing = await db.followUp.findFirst({
   188→    where: { triageRecordId: args.triageRecordId, status: "pending" },
   189→    include: { triageRecord: { select: { county: true, ward: true, classification: true, escalation: true, aggregateTag: true, chpNextAction: true } } },
   190→  });
   191→  if (existing) return toFollowUpDTO(existing);
   192→
   193→  const dueAt = new Date();
   194→  dueAt.setHours(dueAt.getHours() + dueInHours);
   195→
   196→  const created = await db.followUp.create({
   197→    data: {
   198→      triageRecordId: args.triageRecordId,
   199→      chvId: args.chvId,
   200→      dueAt,
   201→    },
   202→    include: { triageRecord: { select: { county: true, ward: true, classification: true, escalation: true, aggregateTag: true, chpNextAction: true } } },
   203→  });
   204→  return toFollowUpDTO(created);
   205→}

   208→export async function getMyFollowUps(
   209→  chvId: string,
   210→  opts: { status?: FollowUpStatus | "all"; limit?: number } = {}
   211→): Promise<FollowUpDTO[]> {
   212→  const status = opts.status ?? "pending";
   213→  const rows = await db.followUp.findMany({

   215→      chvId,
   216→      ...(status !== "all" ? { status } : {}),
   217→    },
   218→    orderBy: [{ status: "asc" }, { dueAt: "asc" }],
   219→    take: opts.limit ?? 50,
   220→    include: {
   221→      triageRecord: { select: { county: true, ward: true, classification: true, escalation: true, aggregateTag: true, chpNextAction: true } },
   222→    },
   223→  });
   224→  return rows.map(toFollowUpDTO);

   227→/** Resolve a follow-up (mark done/missed). Ownership-scoped. */
   228→export async function resolveFollowUp(args: {
   229→  followUpId: string;
   230→  chvId: string;
   231→  status: "done" | "missed";
   232→  resolutionNote?: string;
   233→}): Promise<FollowUpDTO | null> {
   234→  const { followUpId, chvId, status, resolutionNote } = args;
   235→  // Ownership check: only the assigned CHV can resolve.
   236→  const existing = await db.followUp.findUnique({ where: { id: followUpId } });
   237→  if (!existing || existing.chvId !== chvId) return null;
   238→  if (existing.status !== "pending") return null; // already resolved
   239→
   240→  const updated = await db.followUp.update({
   241→    where: { id: followUpId },
   242→    data: {
   243→      status,
   244→      resolvedAt: new Date(),
   245→      resolutionNote: resolutionNote?.trim() || null,
   246→    },
   247→    include: {
   248→      triageRecord: { select: { county: true, ward: true, classification: true, escalation: true, aggregateTag: true, chpNextAction: true } },
   249→    },
   250→  });
   251→  return toFollowUpDTO(updated);
   252→}

   271→/**
   272→ * Returns a CHV's personal aggregate stats (ownership-scoped). Computed from
   273→ * their own records only — the RLS `auth.uid() = submitted_by` equivalent.
   274→ */
   275→export async function getMyStats(submittedById: string): Promise<ChvStats> {
   276→  const now = new Date();
   277→  const last7Start = new Date(now);
   278→  last7Start.setDate(last7Start.getDate() - 7);
   279→  last7Start.setHours(0, 0, 0, 0);
   280→  const prev7Start = new Date(last7Start);
   281→  prev7Start.setDate(prev7Start.getDate() - 7);
   282→
   283→  const [groups, last7Rows, prev7Rows, firstRow, counties] = await Promise.all([
   284→    db.triageRecord.groupBy({
   285→      by: ["classification", "escalation"],
   286→      _count: true,
   287→      where: { submittedById },
   288→    }),
   289→    db.triageRecord.count({
   290→      where: { submittedById, createdAt: { gte: last7Start } },
   291→    }),
   292→    db.triageRecord.count({
   293→      where: {
   294→        submittedById,
   295→        createdAt: { gte: prev7Start, lt: last7Start },
   296→      },
   297→    }),
   298→    db.triageRecord.findFirst({
   299→      where: { submittedById },
   300→      orderBy: { createdAt: "asc" },
   301→      select: { createdAt: true },
   302→    }),
   303→    db.triageRecord.groupBy({
   304→      by: ["county"],
   305→      _count: true,
   306→      where: { submittedById },
   307→    }),
   308→  ]);
   309→
   310→  const stats: ChvStats = {
   311→    total: 0,
   312→    routine: 0,
   313→    needs_followup: 0,
   314→    needs_facility_referral: 0,
   315→    escalation: 0,
   316→    last7d: last7Rows,
   317→    prev7d: prev7Rows,
   318→    firstSubmission: firstRow ? firstRow.createdAt.toISOString() : null,
   319→    countiesCovered: counties.length,
   320→  };
   321→
   322→  for (const g of groups) {
   323→    stats.total += g._count;
   324→    if (g.classification === "routine") stats.routine += g._count;
   325→    if (g.classification === "needs_followup") stats.needs_followup += g._count;
   326→    if (g.classification === "needs_facility_referral")
   327→      stats.needs_facility_referral += g._count;
   328→    if (g.escalation) stats.escalation += g._count;
   329→  }
   330→
   331→  return stats;
   332→}

   357→/**
   358→ * Aggregate by county over the last `days` days. Uses groupBy — never touches
   359→ * observed_indicators text.
   360→ */
   361→export async function getAggregateByCounty(
   362→  days?: number
   363→): Promise<CountyAggregate[]> {
   364→  const since =
   365→    days && days > 0
   366→      ? (() => {
   367→          const d = new Date();
   368→          d.setHours(0, 0, 0, 0);
   369→          d.setDate(d.getDate() - (days - 1));
   370→          return d;
   371→        })()
   372→      : undefined;
   373→  const groups = await db.triageRecord.groupBy({
   374→    by: ["county", "classification", "escalation"],
   375→    _count: true,
   376→    where: since ? { createdAt: { gte: since } } : undefined,
   377→  });
   378→
   379→  const byCounty = new Map<string, CountyAggregate>();
   380→  const ensure = (county: string): CountyAggregate => {
   381→    let a = byCounty.get(county);
   382→    if (!a) {
   383→      a = {
   384→        county,
   385→        routine: 0,
   386→        needs_followup: 0,
   387→        needs_facility_referral: 0,
   388→        escalation: 0,
   389→        total: 0,
   390→      };
   391→      byCounty.set(county, a);
   392→    }
   393→    return a;
   394→  };
   395→
   396→  for (const g of groups) {
   397→    const a = ensure(g.county);
   398→    a.total += g._count;
   399→    if (g.classification === "routine") a.routine += g._count;
   400→    if (g.classification === "needs_followup") a.needs_followup += g._count;
   401→    if (g.classification === "needs_facility_referral") a.needs_facility_referral += g._count;
   402→    if (g.escalation) a.escalation += g._count;
   403→  }
   404→  return Array.from(byCounty.values()).sort((a, b) => b.total - a.total);
   405→}

   407→/**
   408→ * Aggregate by day (last N days). Group by date string.
   409→ */
   410→export async function getAggregateByDay(days = 14): Promise<DailyAggregate[]> {
   411→  const since = new Date();
   412→  since.setHours(0, 0, 0, 0);
   413→  since.setDate(since.getDate() - (days - 1));
   414→
   415→  const rows = await db.triageRecord.findMany({
   416→    where: { createdAt: { gte: since } },
   417→    select: { createdAt: true, classification: true, escalation: true },
   418→  });
   419→
   420→  const byDay = new Map<string, DailyAggregate>();
   421→  for (let i = 0; i < days; i++) {
   422→    const d = new Date(since);
   423→    d.setDate(d.getDate() + i);
   424→    const key = d.toISOString().slice(0, 10);
   425→    byDay.set(key, {
   426→      day: key,
   427→      routine: 0,
   428→      needs_followup: 0,
   429→      needs_facility_referral: 0,
   430→      escalation: 0,
   431→      total: 0,
   432→    });
   433→  }
   434→
   435→  for (const r of rows) {
   436→    const key = r.createdAt.toISOString().slice(0, 10);
   437→    const a = byDay.get(key);
   438→    if (!a) continue;
   439→    a.total++;
   440→    if (r.classification === "routine") a.routine++;
   441→    if (r.classification === "needs_followup") a.needs_followup++;
   442→    if (r.classification === "needs_facility_referral") a.needs_facility_referral++;
   443→    if (r.escalation) a.escalation++;
   444→  }
   445→  return Array.from(byDay.values()).sort((a, b) => a.day.localeCompare(b.day));
   446→}

   448→/**
   449→ * Aggregate by aggregate_tag (top signals) over the last `days` days. Never
   450→ * exposes indicator text.
   451→ */
   452→export async function getAggregateByTag(
   453→  limit = 12,
   454→  days?: number
   455→): Promise<TagAggregate[]> {
   456→  const since =
   457→    days && days > 0
   458→      ? (() => {
   459→          const d = new Date();
   460→          d.setHours(0, 0, 0, 0);
   461→          d.setDate(d.getDate() - (days - 1));
   462→          return d;
   463→        })()
   464→      : undefined;
   465→  const groups = await db.triageRecord.groupBy({
   466→    by: ["aggregateTag"],
   467→    _count: true,
   468→    orderBy: { _count: { aggregateTag: "desc" } },
   469→    take: limit,
   470→    where: since ? { createdAt: { gte: since } } : undefined,
   471→  });
   472→  return groups
   473→    .filter((g) => g.aggregateTag)
   474→    .map((g) => ({ aggregateTag: g.aggregateTag as string, count: g._count }));
   475→}

   516→    total: 0,
   517→    routine: 0,
   518→    needs_followup: 0,
   519→    needs_facility_referral: 0,

   673→/* ------------------------------------------------------------------ */
   674→/* Follow-up completion stats — for dashboard/supervisor metrics.      */
   675→/* ------------------------------------------------------------------ */
   676→
   677→export interface FollowUpStats {
   678→  pending: number;
   679→  done: number;
   680→  missed: number;
   681→  /** Overdue = pending AND past dueAt. */
   682→  overdue: number;
   683→  /** Completion rate = done / (done + missed). 0 if none resolved. */
   684→  completionRate: number;
   685→  total: number;
   686→}
   687→
   688→/**
   689→ * Returns aggregate follow-up stats (de-identified — counts only). Optionally
   690→ * filtered by county and/or a date window. Used by the dashboard's follow-up
   691→ * KPI card and the supervisor roster.
   692→ */
   693→export async function getFollowUpStats(
   694→  opts: { county?: string; days?: number } = {}
   695→): Promise<FollowUpStats> {
   696→  const since = opts.days
   697→    ? (() => {
   698→        const d = new Date();
   699→        d.setHours(0, 0, 0, 0);
   700→        d.setDate(d.getDate() - (opts.days! - 1));
   701→        return d;
   702→      })()
   703→    : undefined;
   704→
   705→  // County filter requires joining through TriageRecord — use the relation.
   706→  const where = {
   707→    ...(since ? { createdAt: { gte: since } } : {}),
   708→    ...(opts.county ? { triageRecord: { county: opts.county } } : {}),
   709→  };
   710→
   711→  const [statusGroups, overdueCount, total] = await Promise.all([
   712→    db.followUp.groupBy({
   713→      by: ["status"],
   714→      _count: true,
   715→      where,
   716→    }),
   717→    db.followUp.count({
   718→      where: {
   719→        ...where,
   720→        status: "pending",
   721→        dueAt: { lt: new Date() },
   722→      },
   723→    }),
   724→    db.followUp.count({ where }),
   725→  ]);
   726→
   727→  const stats: FollowUpStats = {
   728→    pending: 0,
   729→    done: 0,
   730→    missed: 0,
   731→    overdue: overdueCount,
   732→    total,
   733→    completionRate: 0,
   734→  };
   735→  for (const g of statusGroups) {
   736→    if (g.status === "pending") stats.pending = g._count;
   737→    if (g.status === "done") stats.done = g._count;
   738→    if (g.status === "missed") stats.missed = g._count;
   739→  }
   740→  const resolved = stats.done + stats.missed;
   741→  stats.completionRate = resolved > 0 ? Math.round((stats.done / resolved) * 100) : 0;
   742→  return stats;
   743→}

   746→/* Supervisor roster — de-identified per-CHV aggregate for supervisors. */
   747→/* Groups triage records by submitting CHV, returning per-CHV counts    */
   748→/* (never the CHV email, never observation text). A supervisor sees     */
   749→/* activity + load + escalation burden per volunteer.                   */
   750→/* ------------------------------------------------------------------ */
   751→
   752→export interface SupervisorChvRow {
   753→  /** Truncated CHV id — never the email. */
   754→  chvLabel: string;
   755→  county: string;
   756→  ward: string | null;
   757→  total: number;
   758→  routine: number;
   759→  needs_followup: number;
   760→  needs_facility_referral: number;
   761→  escalation: number;
   762→  /** Last 7 days count (for activity recency). */
   763→  last7d: number;
   764→  /** Most recent submission (ISO) — null if none. */
   765→  lastSubmission: string | null;
   766→}
   767→
   768→export interface SupervisorRoster {
   769→  rows: SupervisorChvRow[];
   770→  totals: {
   771→    chvs: number;
   772→    total: number;
   773→    escalations: number;
   774→  };
   775→}

   777→/**
   778→ * Returns a de-identified per-CHV roster for the supervisor view. Groups
   779→ * triage records by submittedById, computes per-CHV aggregates, and labels
   780→ * each row with a truncated id (chv·xxxx) — never the email. Optionally
   781→ * filter by county. Ownership is NOT bypassed at the row level (the
   782→ * supervisor sees aggregates only, never individual observation text).
   783→ *
   784→ * TODO (production): require a supervisor RBAC role. Currently open for demo.
   785→ */
   786→export async function getSupervisorRoster(
   787→  county?: string,
   788→  days = 14
   789→): Promise<SupervisorRoster> {
   790→  const since = (() => {
   791→    const d = new Date();
   792→    d.setHours(0, 0, 0, 0);
   793→    d.setDate(d.getDate() - (days - 1));
   794→    return d;
   795→  })();
   796→  const last7Start = (() => {
   797→    const d = new Date();
   798→    d.setHours(0, 0, 0, 0);
   799→    d.setDate(d.getDate() - 7);
   800→    return d;
   801→  })();
   802→
   803→  const where = {
   804→    createdAt: { gte: since },
   805→    ...(county ? { county } : {}),
   806→  };
   807→
   808→  // Group by submittedById + classification + escalation for per-CHV breakdowns.
   809→  const [groups, chvs, last7Counts, lastSubs] = await Promise.all([
   810→    db.triageRecord.groupBy({
   811→      by: ["submittedById", "classification", "escalation"],
   812→      _count: true,
   813→      where,
   814→    }),
   815→    db.chvUser.findMany({
   816→      where: county ? { county } : undefined,
   817→      select: { id: true, county: true, ward: true },
   818→    }),
   819→    db.triageRecord.groupBy({
   820→      by: ["submittedById"],
   821→      _count: true,
   822→      where: { ...where, createdAt: { gte: last7Start } },
   823→    }),
   824→    db.triageRecord.findMany({
   825→      where,
   826→      distinct: ["submittedById"],
   827→      orderBy: { createdAt: "desc" },
   828→      select: { submittedById: true, createdAt: true },
   829→    }),
   830→  ]);
   831→
   832→  const chvMeta = new Map(chvs.map((c) => [c.id, c]));
   833→  const last7Map = new Map(last7Counts.map((g) => [g.submittedById, g._count]));
   834→  const lastSubMap = new Map(lastSubs.map((r) => [r.submittedById, r.createdAt.toISOString()]));
   835→
   836→  const byChv = new Map<string, SupervisorChvRow>();
   837→  const ensure = (id: string): SupervisorChvRow => {
   838→    let r = byChv.get(id);
   839→    if (!r) {
   840→      const meta = chvMeta.get(id);
   841→      r = {
   842→        chvLabel: `chv·${id.slice(-4)}`,
   843→        county: meta?.county ?? "—",
   844→        ward: meta?.ward ?? null,
   845→        total: 0,
   846→        routine: 0,
   847→        needs_followup: 0,
   848→        needs_facility_referral: 0,
   849→        escalation: 0,
   850→        last7d: last7Map.get(id) ?? 0,
   851→        lastSubmission: lastSubMap.get(id) ?? null,
   852→      };
   853→      byChv.set(id, r);
   854→    }
   855→    return r;
   856→  };
   857→
   858→  let totalAll = 0;
   859→  let totalEsc = 0;
   860→  for (const g of groups) {
   861→    const r = ensure(g.submittedById);
   862→    r.total += g._count;
   863→    totalAll += g._count;
   864→    if (g.classification === "routine") r.routine += g._count;
   865→    if (g.classification === "needs_followup") r.needs_followup += g._count;
   866→    if (g.classification === "needs_facility_referral") r.needs_facility_referral += g._count;
   867→    if (g.escalation) {
   868→      r.escalation += g._count;
   869→      totalEsc += g._count;
   870→    }
   871→  }
   872→
   873→  const rows = Array.from(byChv.values()).sort((a, b) => b.total - a.total);
   874→
   875→  return {
   876→    rows,
   877→    totals: { chvs: rows.length, total: totalAll, escalations: totalEsc },
   878→  };
   879→}
   880→
   881→/**
   882→ * County-scoped dashboard stats — the RBAC path. When a county official is
   883→ * logged in (or a CHV views their own county), we filter every aggregate to
   884→ * a single county. The byCounty array then has at most one row; the daily
   885→ * trend + tags are scoped to that county only.
   886→ */
   887→export async function getDashboardStatsForCounty(
   888→  county: string,
   889→  days = 14
   890→): Promise<DashboardStats & { county: string }> {
   891→  const since = (() => {
   892→    const d = new Date();
   893→    d.setHours(0, 0, 0, 0);
   894→    d.setDate(d.getDate() - (days - 1));
   895→    return d;
   896→  })();
   897→
   898→  const where = {
   899→    county,
   900→    createdAt: { gte: since },
   901→  };
   902→
   903→  const [countyGroups, dayRows, tagGroups, totalsGroups] = await Promise.all([
   904→    db.triageRecord.groupBy({
   905→      by: ["county", "classification", "escalation"],
   906→      _count: true,
   907→      where,
   908→    }),
   909→    db.triageRecord.findMany({
   910→      where,
   911→      select: { createdAt: true, classification: true, escalation: true },
   912→    }),
   913→    db.triageRecord.groupBy({
   914→      by: ["aggregateTag"],
   915→      _count: true,
   916→      orderBy: { _count: { aggregateTag: "desc" } },
   917→      take: 12,
   918→      where,
   919→    }),
   920→    db.triageRecord.groupBy({
   921→      by: ["classification", "escalation"],
   922→      _count: true,
   923→      where,
   924→    }),
   925→  ]);
   926→
   927→  // byCounty — single row for the scoped county
   928→  const countyAgg: CountyAggregate = {
   929→    county,
   930→    routine: 0,
   931→    needs_followup: 0,
   932→    needs_facility_referral: 0,
   933→    escalation: 0,
   934→    total: 0,
   935→  };
   936→  for (const g of countyGroups) {
   937→    countyAgg.total += g._count;
   938→    if (g.classification === "routine") countyAgg.routine += g._count;
   939→    if (g.classification === "needs_followup") countyAgg.needs_followup += g._count;
   940→    if (g.classification === "needs_facility_referral")
   941→      countyAgg.needs_facility_referral += g._count;
   942→    if (g.escalation) countyAgg.escalation += g._count;
   943→  }
   944→  const byCounty = countyAgg.total > 0 ? [countyAgg] : [];
   945→
   946→  // byDay — full N-day ladder, populated from dayRows
   947→  const byDay = new Map<string, DailyAggregate>();
   948→  for (let i = 0; i < days; i++) {
   949→    const d = new Date(since);
   950→    d.setDate(d.getDate() + i);
   951→    const key = d.toISOString().slice(0, 10);
   952→    byDay.set(key, {
   953→      day: key,
   954→      routine: 0,
   955→      needs_followup: 0,
   956→      needs_facility_referral: 0,
   957→      escalation: 0,
   958→      total: 0,
   959→    });
   960→  }
   961→  for (const r of dayRows) {
   962→    const key = r.createdAt.toISOString().slice(0, 10);
   963→    const a = byDay.get(key);
   964→    if (!a) continue;
   965→    a.total++;
   966→    if (r.classification === "routine") a.routine++;
   967→    if (r.classification === "needs_followup") a.needs_followup++;
   968→    if (r.classification === "needs_facility_referral") a.needs_facility_referral++;
   969→    if (r.escalation) a.escalation++;
   970→  }
   971→
   972→  const byTag: TagAggregate[] = tagGroups
   973→    .filter((g) => g.aggregateTag)
   974→    .map((g) => ({ aggregateTag: g.aggregateTag as string, count: g._count }));
   975→
   976→  const totals = {
   977→    total: 0,
   978→    routine: 0,
   979→    needs_followup: 0,
   980→    needs_facility_referral: 0,
   981→    escalation: 0,
   982→    countiesCovered: byCounty.length,
   983→  };
   984→  for (const g of totalsGroups) {
   985→    totals.total += g._count;
   986→    if (g.classification === "routine") totals.routine += g._count;
   987→    if (g.classification === "needs_followup") totals.needs_followup += g._count;
   988→    if (g.classification === "needs_facility_referral")
   989→      totals.needs_facility_referral += g._count;
   990→    if (g.escalation) totals.escalation += g._count;
   991→  }
   992→
   993→  return { byCounty, byDay, byTag, totals, county };
   994→}
